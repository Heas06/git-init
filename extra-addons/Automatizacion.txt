def _find_or_create_lead(self)
        self.ensure_one()
        Lead = self.env[crm.lead].sudo()
        
        # 1. Buscar la etapa Cita Agendada en el CRM
        stage = self.env['crm.lead.stage'].sudo().search([('name', 'ilike', 'Cita Agendada')], limit=1)

        # 2. Buscar si ya existe una oportunidad activa para este cliente
        lead = Lead.search(
            [(partner_id, =, self.partner_id.id), (type, =, opportunity), (probability, , 100)],
            limit=1,
            order=create_date desc
        )

        if lead
            # CASO 1 Si existía en Nuevo Prospecto, mover a Cita Agendada
            if stage
                lead.write({'stage_id' stage.id})
            
            lead.message_post(
                body=self.env._(
                    New salon appointment %(ref)s %(service)s with %(employee)s on %(when)s.,
                    ref=self.name,
                    service=self.service_id.name,
                    employee=self.employee_id.name,
                    when=self.start_display,
                )
            )
            return lead

        # CASO 2 Si agendó directo sin oportunidad previa, crear en Cita Agendada
        vals = {
            name self.env._(%(partner)s - Salon, partner=self.partner_id.name),
            partner_id self.partner_id.id,
            company_id self.company_id.id,
            user_id False,
            type opportunity,
        }
        if stage
            vals[stage_id] = stage.id

        if self.appointment_type_id.crm_team_id
            vals[team_id] = self.appointment_type_id.crm_team_id.id

        return Lead.create(vals)
